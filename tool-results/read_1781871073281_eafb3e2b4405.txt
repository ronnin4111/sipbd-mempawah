    82→/**
    83→ * Classify the user's question type to determine what data to include.
    84→ *
    85→ * IMPORTANT: 'specific' questions load verbose targeted results (all member names),
    86→ * which makes the prompt very large. Counting/summary questions like "berapa jumlah

    88→ * in the data context, not from the full member listings.

    90→function classifyQuestion(message: string): 'specific' | 'personnel' | 'stats' | 'general' | 'comparison' {
    91→  const lower = message.toLowerCase();
    92→
    93→  // ============================================================
    94→  // Counting/summary questions → 'stats' (NOT 'specific'!)
    95→  // These can be answered from the compact data context summaries.
    96→  // Loading targeted results for these just wastes tokens and causes failures.
    97→  // ============================================================
    98→  const countSummaryPatterns = [
    99→    /berapa\s+(jumlah\s+)?kelompok/i,     // "berapa kelompok", "berapa jumlah kelompok"
   100→    /berapa\s+(jumlah\s+)?(rtp|pembudidaya)/i,
   101→    /berapa\s+(jumlah\s+)?kusuka/i,       // "berapa kusuka", "berapa jumlah kusuka"
   102→    /total\s+(kelompok|rtp|pembudidaya|kusuka)/i,
   103→    /jumlah\s+(kelompok|rtp|pembudidaya|kusuka)/i,
   104→    /jumlah\s+anggota/i,                   // "jumlah anggota" = counting, not listing
   105→  ];
   106→  if (countSummaryPatterns.some(p => p.test(lower))) return 'stats';
   107→
   108→  // ============================================================
   109→  // Personnel/pegawai questions → 'personnel' (NOT 'specific'!)
   110→  // These are about org structure, not fishery data.
   111→  // Using compact context + KB focus keeps prompt small and avoids Groq 413.
   112→  // ============================================================
   113→  const personnelPatterns = [
   114→    /jabatan/i, /pegawai/i, /karyawan/i, /staff/i, /staf/i,
   115→    /pangkat/i, /golongan/i, /unit\s+kerja/i,
   116→    /kepala\s+(dinas|bidang|seksi|sub)/i, /kabid/i, /kasubid/i, /kasi/i,
   117→    /eselon/i, /nip/i,
   118→    // Person-name patterns: "siapa hasto", "apa jabatan X", "nama Arifin"
   119→    /siapa\s+[A-Z]/i, /jabatan\s+[A-Z]/i,
   120→    // Personnel listing patterns: "siapa saja pegawai", "daftar pegawai"
   121→    /siapa\s+saja\s+(pegawai|karyawan|staf|dinas)/i,
   122→    /daftar\s+(pegawai|karyawan|staf|dinas)/i,
   123→    /sebutkan\s+(pegawai|karyawan|staf|dinas)/i,
   124→    /tampilkan\s+(pegawai|karyawan|staf|dinas)/i,
   125→  ];
   126→  // Also check for person names (capitalized multi-word entities)
   127→  const hasPersonName = /\b([A-Z][a-z]+(?:\s+[A-Z][a-z]+)+)\b/.test(message);
   128→  // Check if question is about org structure even without person name
   129→  const isOrgQuestion = /kepala\s+(dinas|bidang|seksi|sub|dpa)|struktur\s+organisasi/i.test(lower);
   130→  // Check if question mentions personnel-related words with listing intent
   131→  const isPersonnelListing = /pegawai|karyawan|staf|dinas|jabatan|golongan|pangkat/i.test(lower) &&
   132→    /siapa\s+saja|daftar|sebutkan|tampilkan|semua|seluruh|nama.*nama/i.test(lower);
   133→  if (personnelPatterns.some(p => p.test(lower)) || isOrgQuestion || isPersonnelListing ||
   134→      (hasPersonName && /jabatan|pegawai|staf|golongan|pangkat|dinas|bidang/i.test(lower))) {
   135→    return 'personnel';
   136→  }
   137→
   138→  // Specific group/farmer questions — these need individual member details
   139→  const specificPatterns = [
   140→    /anggota\s+kelompok/i, /pembudidaya\s+\w+/i,
   141→    /siapa\s+saja/i, /daftar\s+anggota/i, /nama\s+kelompok/i,
   142→    /semua\s+kelompok/i, /seluruh\s+kelompok/i,
   143→    /tampilkan\s+semua/i, /tampilkan\s+seluruh/i, /daftar\s+kelompok/i,
   144→    // NOTE: /siapa\s+saja/ is here as a fallback — it's checked AFTER
   145→    // the personnel patterns, so "siapa saja pegawai" will match personnel first
   146→    /kelompok\s+(pembenih|pembenihan|pembesaran)/i,
   147→    // NOTE: Removed standalone /pembenih|pembenihan|pembesaran/i — too broad.
   148→    // It matched ANY mention of pembesaran, even in counting questions like
   149→    // "berapa produksi pembesaran" which should be 'stats'.
   150→    /nama\s+anggota/i, /anggota\s+dari/i,
   151→    // "kelompok [nama]" — specific group by name (e.g. "kelompok kawan sejati")
   152→    // Must come AFTER /kelompok\s+(pembenih|pembenihan|pembesaran)/i so business types match first
   153→    // Exclude common function words: "di", "yang", "ini", "itu", "dari", "ke", "ada", "apa"
   154→    /kelompok\s+(?!di\b|yang\b|ini\b|itu\b|dari\b|ke\b|ada\b|apa\b|dengan\b|untuk\b|pada\b|semua\b|seluruh\b)\w+/i,
   155→    // "berapa orang/anggota" + specific group name context
   156→    // e.g. "kelompok kawan sejati mempunyai anggota berapa orang"
   157→    /berapa\s+(orang|anggota|member)/i,
   158→    // KUSUKA-specific patterns
   159→    /kusuka/i, /kartu\s+usaha/i, /registrasi/i, /pendaftaran/i,
   160→    /perorangan/i, /pembenih\s+perorangan/i, /nama\s+pembudidaya/i,
   161→    /alamat\s+pembudidaya/i,
   162→    /status\s+kusuka/i, /draf\s+kusuka/i, /valid\s+kusuka/i,
   163→    // "bidang" alone (without personnel context) is a sector question
   164→    /bidang/i,
   165→    /nama\s+\w+/i,  // "nama Roni" pattern — still specific for fishery names
   166→  ];
   167→  if (specificPatterns.some(p => p.test(lower))) return 'specific';
   168→
   169→  // Comparison questions — multi-year or multi-region comparisons
   170→  const comparisonPatterns = [
   171→    /bandingkan/i, /perbandingan/i, /perbedaan/i, /compare/i,
   172→    /vs\.?/i, /\d{4}\s*(vs|dengan|dan|terhadap)\s*\d{4}/i,
   173→    /\d{4}\s*-\s*\d{4}/,  // "2024-2025" range
   174→  ];
   175→  if (comparisonPatterns.some(p => p.test(lower))) return 'comparison';
   176→
   177→  // Stats/production/trend questions (but NOT kelompok listing questions)
   178→  const statsPatterns = [
   179→    /produksi/i, /tren/i, /statistik/i,
   180→    /kecamatan.*tinggi/i, /kecamatan.*rendah/i, /perbandingan/i,
   181→    /rtp/i, /cpib/i, /cbib/i, /pencapaian/i, /target/i,
   182→    /naik/i, /turun/i, /kenaikan/i, /penurunan/i, /pertumbuhan/i,
   183→    /jumlah\s+(pembudidaya|rtp|produksi)/i, /total\s+(pembudidaya|rtp|produksi)/i,
   184→  ];
   185→  if (statsPatterns.some(p => p.test(lower))) return 'stats';
   186→
   187→  return 'general';
   188→}

   190→/**
   191→ * Extract potential search keywords from the user's message.
   192→ * Includes person names as multi-word entities.
   193→ */
   194→function extractSearchTerms(message: string): string[] {
   195→  const terms: string[] = [];
   196→
   197→  // Skip business type queries — they're handled by the data context breakdown
   198→  const isBizTypeQuery = /kelompok\s+(pembenih|pembenihan|pembesaran)/i.test(message) ||
   199→    /^(berapa\s+)?(jumlah\s+)?(kelompok\s+)?(pembenih|pembenihan|pembesaran)/i.test(message);
   200→  if (isBizTypeQuery) {
   201→    // For business type queries, no search terms needed — data context has the breakdown
   202→    return [];
   203→  }
   204→
   205→  // Extract multi-word entities (person names like "Roni Irama")
   206→  const entityPattern = /\b([A-Z][a-z]+(?:\s+[A-Z][a-z]+)+)\b/g;
   207→  let entityMatch;
   208→  while ((entityMatch = entityPattern.exec(message)) !== null) {
   209→    const entity = entityMatch[1].trim();
   210→    if (entity.length > 3) {
   211→      terms.push(entity);
   212→    }
   213→  }
   214→
   215→  const patterns = [
   216→    /anggota\s+kelompok\s+([^\?,\.\!]+)/i,
   217→    /kelompok\s+([^\?,\.\!]+)/i,
   218→    /pembudidaya\s+([^\?,\.\!]+)/i,
   219→    /kelompok\s+(\w+(?:\s+\w+)*)/i,
   220→    /grup\s+([^\?,\.\!]+)/i,
   221→    /["']([^"']+)["']/,
   222→    /jabatan\s+([^\?,\.\!]+)/i,
   223→    /nama\s+([^\?,\.\!]+)/i,
   224→    /pegawai\s+([^\?,\.\!]+)/i,
   225→  ];
   226→
   227→  for (const pattern of patterns) {
   228→    const match = message.match(pattern);
   229→    if (match?.[1] && match[1].trim().length > 2) {
   230→      terms.push(match[1].trim());
   231→    }
   232→  }
   233→
   234→  // Proper nouns (capitalized words not in common set)
   235→  const commonWords = new Set([
   236→    'berapa', 'jumlah', 'anggota', 'kelompok', 'pembudidaya', 'siapa',
   237→    'saja', 'apa', 'dimana', 'kapan', 'bagaimana', 'mengapa', 'yang',
   238→    'di', 'ke', 'dari', 'dengan', 'untuk', 'pada', 'adalah', 'ini',
   239→    'itu', 'dan', 'atau', 'tetapi', 'karena', 'jika', 'kalau', 'bisa',
   240→    'ada', 'tidak', 'sudah', 'belum', 'akan', 'dapat', 'harus', 'perlu',
   241→    'semua', 'setiap', 'lain', 'lainnya', 'total', 'data', 'produksi',
   242→    'tahun', 'kecamatan', 'desa', 'ikan', 'jenis', 'usaha', 'rtp',
   243→    'kusuka', 'cpib', 'cbib', 'halo', 'hai', 'hello', 'hi', 'tolong',
   244→    'bantu', 'jelaskan', 'sebutkan', 'daftar', 'informasi', 'tentang',
   245→    'kabupaten', 'mempawah', 'dinas', 'perikanan', 'budidaya',
   246→    'seluruh', 'tampilkan', 'sebutkan',
   247→    'pembenih', 'pembenihan', 'pembesaran',  // handled specially, not as search terms
   248→    'registrasi', 'pendaftaran', 'perorangan', 'kartu',  // KUSUKA keywords, not search terms
   249→    'jabatan', 'pegawai', 'karyawan', 'staff', 'staf',  // question context, not search terms
   250→    'pangkat', 'golongan', 'unit', 'bidang',  // question context
   251→  ]);
   252→
   253→  const words = message.split(/\s+/);
   254→  for (const word of words) {
   255→    const clean = word.replace(/[^\w]/g, '');
   256→    if (clean.length > 2 && clean[0] === clean[0].toUpperCase() && !commonWords.has(clean.toLowerCase())) {
   257→      terms.push(clean);
   258→    }
   259→  }
   260→
   261→  return [...new Set(terms)];
   262→}

   284→/**
   285→ * In-memory cache for dynamic lists from DB, with 5-minute TTL.
   286→ * Avoids hitting the database on every request while still
   287→ * staying up-to-date when new kecamatan/fish types are added.
   288→ */
   289→interface CachedList {
   290→  values: string[];
   291→  updatedAt: number;
   292→}
   293→const CACHE_TTL_MS = 5 * 60 * 1000; // 5 minutes
   294→let kecamatanCache: CachedList | null = null;
   295→let fishTypeCache: CachedList | null = null;
   296→let yearsCache: CachedList | null = null;
   297→
   298→/**
   299→ * Get distinct kecamatan names from the database.
   300→ * Results are cached in memory for 5 minutes.
   301→ * Falls back to FALLBACK_KECAMATAN if the DB query fails.
   302→ */
   303→async function getDynamicKecamatanList(): Promise<string[]> {
   304→  if (kecamatanCache && Date.now() - kecamatanCache.updatedAt < CACHE_TTL_MS) {
   305→    return kecamatanCache.values;
   306→  }
   307→  try {
   308→    const result = await db.fishFarm.findMany({
   309→      select: { kecamatan: true },
   310→      distinct: ['kecamatan'],
   311→      orderBy: { kecamatan: 'asc' },
   312→    });
   313→    const values = result.map(r => r.kecamatan).filter(Boolean);
   314→    if (values.length > 0) {
   315→      kecamatanCache = { values, updatedAt: Date.now() };
   316→      console.log(`[parseQuestionContext] Loaded ${values.length} kecamatan from DB:`, values.join(', '));
   317→      return values;
   318→    }
   319→  } catch (err) {
   320→    console.error('[parseQuestionContext] DB query for kecamatan failed, using fallback:', err);
   321→  }
   322→  return FALLBACK_KECAMATAN;
   323→}

   357→async function getAvailableYears(): Promise<number[]> {
   358→  if (yearsCache && Date.now() - yearsCache.updatedAt < CACHE_TTL_MS) {
   359→    return yearsCache.values.map(Number);
   360→  }
   361→  try {
   362→    const result = await db.fishFarm.findMany({
   363→      select: { year: true },
   364→      distinct: ['year'],
   365→      orderBy: { year: 'desc' },
   366→    });
   367→    const values = result.map(r => r.year).filter((y): y is number => typeof y === 'number');
   368→    if (values.length > 0) {
   369→      yearsCache = { values: values.map(String), updatedAt: Date.now() };
   370→      console.log(`[getAvailableYears] Loaded ${values.length} years from DB:`, values.join(', '));
   371→      return values;
   372→    }
   373→  } catch {
   374→    // fall through
   375→  }
   376→  return [];
   377→}

   379→/**
   380→ * Parsed question context — what the user is asking about,
   381→ * extracted from the natural language question itself
   382→ * (independent of UI filter state).
   383→ */
   384→interface QuestionContext {
   385→  years: number[];
   386→  kecamatan: string[];
   387→  businessType: string[];
   388→  fishType: string[];
   389→  desa: string[];
   390→  isComparison: boolean;
   391→}
   392→
   393→/**
   394→ * Parse the user's question to extract contextual information
   395→ * like years, kecamatan, fish types, etc.
   396→ * This is used to override UI filters so the AI fetches
   397→ * the right data regardless of what the UI shows.
   398→ *
   399→ * Uses dynamic lists from the database (cached for 5 min)
   400→ * to avoid hardcoding kecamatan/fish types that may be wrong
   401→ * (e.g. "Weda Selatan" which is in Halmahera, not Mempawah).
   402→ */
   403→async function parseQuestionContext(message: string): Promise<QuestionContext> {
   404→  const ctx: QuestionContext = {
   405→    years: [],
   406→    kecamatan: [],
   407→    businessType: [],
   408→    fishType: [],
   409→    desa: [],
   410→    isComparison: false,
   411→  };
   412→  const lower = message.toLowerCase();
   413→  const currentYear = new Date().getFullYear();
   414→
   415→  // ===== Parse years =====
   416→  // "tahun 2025", "data 2024", "tahun 2024 dan 2025"
   417→  const explicitYearPattern = /(?:tahun|data|tahun\s+ini|tahun\s+lalu|tahun\s+kemarin)?\s*(\d{4})(?:\s*(?:dan|,|&)?\s*(\d{4}))*/gi;
   418→  const explicitYears = new Set<number>();
   419→  let yearMatch: RegExpExecArray | null;
   420→  while ((yearMatch = explicitYearPattern.exec(lower)) !== null) {
   421→    for (let i = 1; i < yearMatch.length; i++) {
   422→      if (yearMatch[i]) {
   423→        const y = parseInt(yearMatch[i], 10);
   424→        if (y >= 2020 && y <= currentYear + 1) {
   425→          explicitYears.add(y);
   426→        }
   427→      }
   428→    }
   429→  }
   430→
   431→  // "2024-2026" range pattern
   432→  const rangePattern = /(\d{4})\s*-\s*(\d{4})/;
   433→  const rangeMatch = lower.match(rangePattern);
   434→  if (rangeMatch) {
   435→    const startY = parseInt(rangeMatch[1], 10);
   436→    const endY = parseInt(rangeMatch[2], 10);
   437→    if (startY < endY && endY - startY <= 10) {
   438→      for (let y = startY; y <= endY; y++) {
   439→        explicitYears.add(y);
   440→      }
   441→    }
   442→  }
   443→
   444→  // Relative year patterns
   445→  if (/tahun\s+lalu|tahun\s+kemarin/i.test(lower)) {
   446→    explicitYears.add(currentYear - 1);
   447→  }
   448→  if (/tahun\s+ini|tahun\s+sekarang|tahun\s+berjalan/i.test(lower)) {
   449→    explicitYears.add(currentYear);
   450→  }
   451→
   452→  // "N tahun terakhir" pattern
   453→  const nYearPattern = /(\d+)\s+tahun\s+terakhir/i;
   454→  const nYearMatch = lower.match(nYearPattern);
   455→  if (nYearMatch) {
   456→    const n = parseInt(nYearMatch[1], 10);
   457→    if (n >= 1 && n <= 10) {
   458→      for (let i = 0; i < n; i++) {
   459→        explicitYears.add(currentYear - i);
   460→      }
   461→    }
   462→  }
   463→
   464→  // "bandingkan 2024 vs 2025" / "perbandingan 2024 2025"
   465→  const vsPattern = /(\d{4})\s*(?:vs|dengan|dan|terhadap)\s*(\d{4})/i;
   466→  const vsMatch = lower.match(vsPattern);
   467→  if (vsMatch) {
   468→    const y1 = parseInt(vsMatch[1], 10);
   469→    const y2 = parseInt(vsMatch[2], 10);
   470→    if (y1 >= 2020 && y1 <= currentYear + 1) explicitYears.add(y1);
   471→    if (y2 >= 2020 && y2 <= currentYear + 1) explicitYears.add(y2);
   472→    ctx.isComparison = true;
   473→  }
   474→
   475→  // Check comparison keywords
   476→  if (/bandingkan|perbandingan|perbedaan|compare/i.test(lower)) {
   477→    ctx.isComparison = true;
   478→  }
   479→
   480→  ctx.years = [...explicitYears].sort((a, b) => a - b);
   481→
   482→  // ===== Parse kecamatan =====
   483→  const dynamicKecamatan = await getDynamicKecamatanList();
   484→  for (const kec of dynamicKecamatan) {
   485→    if (lower.includes(kec.toLowerCase())) {
   486→      ctx.kecamatan.push(kec);
   487→    }
   488→  }
   489→  // Fuzzy match: also check partial kecamatan names in the question
   490→  // Build partial patterns dynamically from DB kecamatan names
   491→  const kecPartialPatterns: Record<string, string[]> = {};
   492→  for (const kec of dynamicKecamatan) {
   493→    const patterns = [kec.toLowerCase()];
   494→    // For multi-word kecamatan, add the last word as a partial pattern
   495→    const parts = kec.split(' ');
   496→    if (parts.length > 1) {
   497→      for (const part of parts) {
   498→        if (part.length > 3) patterns.push(part.toLowerCase());
   499→      }
   500→    }
   501→    kecPartialPatterns[kec] = patterns;
   502→  }
   503→  for (const [kec, patterns] of Object.entries(kecPartialPatterns)) {
   504→    if (ctx.kecamatan.includes(kec)) continue; // already matched
   505→    for (const p of patterns) {
   506→      if (lower.includes(p) && p.length > 3) { // avoid matching very short partials like 'hulu'
   507→        ctx.kecamatan.push(kec);
   508→        break;
   509→      }
   510→    }
   511→  }
   512→
   513→  // ===== Parse business type =====
   514→  if (/pembenihan|pembenih/i.test(lower)) {
   515→    ctx.businessType.push('Pembenihan');
   516→  }
   517→  if (/pembesaran/i.test(lower)) {
   518→    ctx.businessType.push('Pembesaran');
   519→  }
   520→
   521→  // ===== Parse fish type =====
   522→  const dynamicFishTypes = await getDynamicFishTypeList();
   523→  for (const fish of dynamicFishTypes) {
   524→    if (lower.includes(fish.toLowerCase())) {
   525→      ctx.fishType.push(fish);
   526→    }
   527→  }
   528→
   529→  // ===== Parse desa (capitalized words that might be desa names) =====
   530→  const skipWords = new Set([
   531→    ...dynamicKecamatan.map(k => k.toLowerCase()),
   532→    'mempawah', 'kabupaten', 'dinas', 'perikanan', 'budidaya',
   533→    'kelompok', 'pembudidaya', 'anggota', 'jumlah', 'berapa',
   534→    'data', 'tahun', 'desa', 'kecamatan', 'ikan', 'jenis',
   535→    'usaha', 'rtp', 'kusuka', 'produksi', 'pembesaran', 'pembenihan',
   536→    ...dynamicFishTypes.map(f => f.toLowerCase()),
   537→  ]);
   538→  const words = message.split(/\s+/);
   539→  for (const word of words) {
   540→    const clean = word.replace(/[^\w]/g, '');
   541→    if (clean.length > 2 && clean[0] === clean[0].toUpperCase() && !skipWords.has(clean.toLowerCase())) {
   542→      ctx.desa.push(clean);
   543→    }
   544→  }
   545→
   546→  return ctx;
   547→}
   548→
   549→/**
   550→ * Resolve effective filters by merging question context with UI filters.
   551→ * Question context takes PRIORITY over UI filters.
   552→ * If user mentions "tahun 2025" → use 2025, regardless of UI filter.
   553→ * If user doesn't mention any year → fall back to UI filter, then latest 2 years from DB.
   554→ */
   555→async function resolveEffectiveFilters(
   556→  questionCtx: QuestionContext,
   557→  uiFilters: {
   558→    years: string[];
   559→    kecamatan: string[];
   560→    desa: string[];
   561→    fishType: string[];
   562→    containerType: string[];
   563→    businessType: string[];
   564→  }
   565→): Promise<{
   566→  years: string[];
   567→  kecamatan: string[];
   568→  desa: string[];
   569→  fishType: string[];
   570→  containerType: string[];
   571→  businessType: string[];
   572→}> {
   573→  let resolvedYears: string[];
   574→
   575→  if (questionCtx.years.length > 0) {
   576→    // Question context has years → use them
   577→    resolvedYears = questionCtx.years.map(String);
   578→  } else if (uiFilters.years.length > 0) {
   579→    // UI filter has years → use them
   580→    resolvedYears = uiFilters.years;
   581→  } else {
   582→    // Neither has years → use latest 2 years from database
   583→    // This prevents defaulting to just current year which may have incomplete data
   584→    const availableYears = await getAvailableYears();
   585→    if (availableYears.length > 0) {
   586→      resolvedYears = availableYears.slice(0, 2).map(String);
   587→    } else {
   588→      // Fallback to current year if DB query fails
   589→      resolvedYears = [String(new Date().getFullYear())];
   590→    }
   591→  }
   592→
   593→  const resolved = {
   594→    years: resolvedYears,
   595→    // Kecamatan: question context > UI filter
   596→    kecamatan: questionCtx.kecamatan.length > 0
   597→      ? questionCtx.kecamatan
   598→      : uiFilters.kecamatan,
   599→    // Desa: question context > UI filter
   600→    desa: questionCtx.desa.length > 0
   601→      ? questionCtx.desa
   602→      : uiFilters.desa,
   603→    // Fish type: question context > UI filter
   604→    fishType: questionCtx.fishType.length > 0
   605→      ? questionCtx.fishType
   606→      : uiFilters.fishType,
   607→    // Container type: only from UI filter (rarely mentioned in questions)
   608→    containerType: uiFilters.containerType,
   609→    // Business type: question context > UI filter
   610→    businessType: questionCtx.businessType.length > 0
   611→      ? questionCtx.businessType
   612→      : uiFilters.businessType,
   613→  };
   614→
   615→  return resolved;
   616→}

   619→ * Build a human-readable description of the auto-detected question context,
   620→ * to include in the data context header so the AI knows what scope was fetched.
   621→ */
   622→function buildContextHeader(questionCtx: QuestionContext): string {
   623→  const parts: string[] = [];
   624→  if (questionCtx.years.length > 0) {
   625→    parts.push(`Tahun=${questionCtx.years.join(',')}`);
   626→  }
   627→  if (questionCtx.kecamatan.length > 0) {
   628→    parts.push(`Kecamatan=${questionCtx.kecamatan.join(',')}`);
   629→  }
   630→  if (questionCtx.businessType.length > 0) {
   631→    parts.push(`JenisUsaha=${questionCtx.businessType.join(',')}`);
   632→  }
   633→  if (questionCtx.fishType.length > 0) {
   634→    parts.push(`JenisIkan=${questionCtx.fishType.join(',')}`);
   635→  }
   636→  if (questionCtx.desa.length > 0) {
   637→    parts.push(`Desa=${questionCtx.desa.join(',')}`);
   638→  }
   639→  if (questionCtx.isComparison) {
   640→    parts.push('Mode=Perbandingan');
   641→  }
   642→  return parts.length > 0
   643→    ? `Konteks pertanyaan terdeteksi: ${parts.join(', ')}`
   644→    : '';
   645→}

   647→/**
   648→ * Fetch KUSUKA registration data context — summary of all registrations.
   649→ * Provides aggregate statistics for AI to answer KUSUKA-related questions.
   650→ */
   651→async function fetchKusukaDataContext(): Promise<string> {
   652→  try {
   653→    const registrations = await db.kusukaRegistration.findMany();
   654→
   655→    if (registrations.length === 0) {
   656→      return '\n=== DATA KUSUKA (Registrasi Perorangan) ===\nTidak ada data registrasi KUSUKA.';
   657→    }
   658→
   659→    const total = registrations.length;
   660→
   661→    // Count per status
   662→    const statusCount = new Map<string, number>();
   663→    for (const r of registrations) {
   664→      const status = r.statusKusuka || '-';
   665→      statusCount.set(status, (statusCount.get(status) || 0) + 1);
   666→    }
   667→    const statusLines = [...statusCount.entries()]
   668→      .sort(([, a], [, b]) => b - a)
   669→      .map(([s, c]) => `${s}: ${c}`)
   670→      .join(', ');
   671→
   672→    // Count per kecamatan
   673→    const kecCount = new Map<string, number>();
   674→    for (const r of registrations) {
   675→      const kec = r.kecamatan || '-';
   676→      kecCount.set(kec, (kecCount.get(kec) || 0) + 1);
   677→    }
   678→    const kecLines = [...kecCount.entries()]
   679→      .sort(([, a], [, b]) => b - a)
   680→      .map(([k, c]) => `${k}: ${c}`)
   681→      .join(', ');
   682→
   683→    // Count per profesi utama
   684→    const profesiCount = new Map<string, number>();
   685→    for (const r of registrations) {
   686→      const p = r.profesiUtama || '-';
   687→      profesiCount.set(p, (profesiCount.get(p) || 0) + 1);
   688→    }
   689→    const profesiLines = [...profesiCount.entries()]
   690→      .sort(([, a], [, b]) => b - a)
   691→      .map(([p, c]) => `${p}: ${c}`)
   692→      .join(', ');
   693→
   694→    // Count per bentuk usaha
   695→    const bentukCount = new Map<string, number>();
   696→    for (const r of registrations) {
   697→      const b = r.bentukUsaha || '-';
   698→      bentukCount.set(b, (bentukCount.get(b) || 0) + 1);
   699→    }
   700→    const bentukLines = [...bentukCount.entries()]
   701→      .sort(([, a], [, b]) => b - a)
   702→      .map(([b, c]) => `${b}: ${c}`)
   703→      .join(', ');
   704→
   705→    // Count with valid KUSUKA card number (16 digits)
   706→    const validKusukaCard = registrations.filter(r => /^\d{16}$/.test((r.noKusuka || '').trim())).length;
   707→
   708→    // Count with kelompok vs without kelompok (independent)
   709→    const withKelompok = registrations.filter(r => r.namaKelompok && r.namaKelompok.trim() !== '').length;
   710→    const withoutKelompok = total - withKelompok;
   711→
   712→    // Count per desa (grouped by kecamatan)
   713→    const desaCount = new Map<string, Map<string, number>>(); // kecamatan -> desa -> count
   714→    for (const r of registrations) {
   715→      const kec = r.kecamatan || '-';
   716→      const desa = r.kelDesa || '-';
   717→      if (!desaCount.has(kec)) desaCount.set(kec, new Map());
   718→      const desaMap = desaCount.get(kec)!;
   719→      desaMap.set(desa, (desaMap.get(desa) || 0) + 1);
   720→    }
   721→    const desaLines = [...desaCount.entries()]
   722→      .sort(([a], [b]) => a.localeCompare(b))
   723→      .map(([kec, desas]) => {
   724→        const desaDetail = [...desas.entries()]
   725→          .sort(([, a], [, b]) => b - a)
   726→          .map(([desa, count]) => `${desa}:${count}`)
   727→          .join(', ');
   728→        return `${kec} → ${desaDetail}`;
   729→      })
   730→      .join('\n');
   731→
   732→    // List unique kelompok with member counts
   733→    const kelompokMap = new Map<string, number>();
   734→    for (const r of registrations) {
   735→      if (r.namaKelompok && r.namaKelompok.trim()) {
   736→        const k = r.namaKelompok.trim();
   737→        kelompokMap.set(k, (kelompokMap.get(k) || 0) + 1);
   738→      }
   739→    }
   740→    const kelompokLines = [...kelompokMap.entries()]
   741→      .sort(([, a], [, b]) => b - a)
   742→      .slice(0, 30) // Limit to top 30 kelompok to keep compact
   743→      .map(([k, c]) => `${k} (${c})`)
   744→      .join(', ');
   745→    const kelompokExtra = kelompokMap.size > 30 ? ` ...dan ${kelompokMap.size - 30} lagi` : '';
   746→
   747→    const dataContext = `\n=== DATA KUSUKA (Registrasi Perorangan) ===
   748→Total registran: ${total}
   749→Per status: ${statusLines}
   750→Per kecamatan: ${kecLines}
   751→Per profesi utama: ${profesiLines}
   752→Per bentuk usaha: ${bentukLines}
   753→No.KUSUKA valid (16 digit): ${validKusukaCard}
   754→Dengan kelompok: ${withKelompok}, Mandiri (tanpa kelompok): ${withoutKelompok}
   755→Kelompok (${kelompokMap.size}): ${kelompokLines}${kelompokExtra}
   756→
   757→Per desa (kecamatan → desa:jumlah):
   758→${desaLines}`;
   759→
   760→    return dataContext;
   761→  } catch (error) {
   762→    console.error('Failed to fetch KUSUKA data context:', error);
   763→    return '\n=== DATA KUSUKA ===\nGagal memuat data KUSUKA.';
   764→  }
   765→}

   767→/**
   768→ * Fetch targeted KUSUKA registration results based on search terms.
   769→ * Returns full details for matching registrants.

   771→async function fetchKusukaTargetedResults(searchTerms: string[], questionType: string): Promise<string> {
   772→  try {
   773→    const registrations = await db.kusukaRegistration.findMany();
   774→
   775→    if (registrations.length === 0) {
   776→      return '';
   777→    }
   778→
   779→    // If no search terms but it's a specific KUSUKA question, return summary listing
   780→    if (searchTerms.length === 0) {
   781→      if (questionType === 'specific') {
   782→        // Return a compact listing of all registrants (name, kec, status)
   783→        const lines = registrations
   784→          .sort((a, b) => a.nama.localeCompare(b.nama))
   785→          .slice(0, 50) // Limit to keep prompt manageable
   786→          .map((r, i) => `${i + 1}. ${r.nama} | Kec:${r.kecamatan} | Desa:${r.kelDesa} | ${r.bentukUsaha} | ${r.profesiUtama} | Status:${r.statusKusuka}`)
   787→          .join('\n');
   788→        const extra = registrations.length > 50 ? `\n...dan ${registrations.length - 50} registran lainnya` : '';
   789→        return `\n=== DAFTAR REGISTRAN KUSUKA (${registrations.length} orang) ===\n${lines}${extra}`;
   790→      }
   791→      return '';
   792→    }
   793→
   794→    // Search for matching registrants
   795→    const found: string[] = [];
   796→    const matchedIds = new Set<string>();
   797→
   798→    for (const term of searchTerms) {
   799→      const q = term.toLowerCase();
   800→
   801→      for (const r of registrations) {
   802→        if (matchedIds.has(r.id)) continue;
   803→
   804→        if (
   805→          (r.nama && r.nama.toLowerCase().includes(q)) ||
   806→          (r.kecamatan && r.kecamatan.toLowerCase().includes(q)) ||
   807→          (r.kelDesa && r.kelDesa.toLowerCase().includes(q)) ||
   808→          (r.namaKelompok && r.namaKelompok.toLowerCase().includes(q)) ||
   809→          (r.noKusuka && r.noKusuka.toLowerCase().includes(q))
   810→        ) {
   811→          matchedIds.add(r.id);
   812→          // Return full details for matched registrants
   813→          const kelStr = r.namaKelompok ? `Kel:${r.namaKelompok}` : 'Mandiri';
   814→          const noKusukaStr = r.noKusuka ? `No.KUSUKA:${r.noKusuka}` : 'No.KUSUKA:-';
   815→          found.push(`${r.nama} | Kec:${r.kecamatan} | Desa:${r.kelDesa} | ${kelStr} | ${r.bentukUsaha} | ${r.profesiUtama} | ${noKusukaStr} | Alamat:${r.alamat || '-'} | Status:${r.statusKusuka}`);
   816→        }
   817→      }
   818→    }
   819→
   820→    if (found.length === 0) {
   821→      return `\nHASIL PENCARIAN KUSUKA: Tidak ditemukan untuk "${searchTerms.join(', ')}". Coba periksa ejaan.`;
   822→    }
   823→
   824→    return `\n=== HASIL PENCARIAN KUSUKA (${found.length} ditemukan) ===\n${found.join('\n')}`;
   825→  } catch (error) {
   826→    console.error('Failed to fetch KUSUKA targeted results:', error);
   827→    return '';
   828→  }
   829→}

   831→/**
   832→ * Fetch COMPACT data context — summary only, no individual group details.

   834→ * The AI can still answer basic questions like "what data do you have?"

   836→async function fetchCompactDataContext(year?: number): Promise<{
   837→  dataContext: string;
   838→  totalGroups: number;
   839→}> {
   840→  try {
   841→    let targetYear = year;
   842→    if (!targetYear) {
   843→      const availableYears = await getAvailableYears();
   844→      targetYear = availableYears.length > 0 ? availableYears[0] : new Date().getFullYear();
   845→    }
   846→    const records = await db.fishFarm.findMany({ where: { year: targetYear } });
   847→
   848→    if (records.length === 0) {
   849→      return {
   850→        dataContext: '\n=== DATA CONTEXT (Ringkasan) ===\nBelum ada data tahun ini.',
   851→        totalGroups: 0,
   852→      };
   853→    }
   854→
   855→    // Quick counts
   856→    const groupKeys = new Set<string>();
   857→    const kecSet = new Set<string>();
   858→    const fishSet = new Set<string>();
   859→    const bizSet = new Set<string>();
   860→    let totalFarmers = 0;
   861→

  1663→

  1680→
  1681→  // Production by fish type (compact)
  1682→  if (s.productionByFishType && typeof s.productionByFishType === 'object') {
  1683→    const fishData = s.productionByFishType as Record<string, number>;
  1684→    lines.push('Per jenis ikan: ' + Object.entries(fishData)
  1685→      .map(([k, v]) => `${k}: ${Number(v).toLocaleString('id-ID')}`)
