    97→

    99→    scrollToBottom();
   100→  }, [messages, scrollToBottom]);
   101→
   102→  useEffect(() => {
   103→    if (isOpen && inputRef.current && !showConfig) {

   116→          password: configPassword,
   117→          geminiApiKey: configGeminiKey || undefined,

   152→          dbConnection: { ok: false, error: 'Test endpoint failed' },

   154→        });

   157→      setTestResult({

   159→        groq: { keyFound: false, keySource: 'none', keyHint: null, testResult: 'failed', latencyMs: 0, error: 'Network error' },
   160→        dbConnection: { ok: false, error: 'Network error' },

   173→      content: text.trim(),

   185→        content: m.content,

   190→        ? {
   191→            periodLabel: stats.periodLabel,
   192→            currentYear: stats.currentYear,
   193→            pembesaranProduction: stats.pembesaranProduction,

'⚠️ **Provider AI tidak tersedia.**\n\nZ.AI sedang tidak dapat diakses, dan API key Gemini/Groq belum dikonfigurasi.\n\nKlik ikon ⚙️ di header chat untuk:\n- Cek status Z.AI\n- Atur API key Gemini/Groq sebagai fallback\n\n**Gratis & mudah:**\n- Gemini: https://aistudio.google.com/apikey\n- Groq: https://console.groq.com'

   264→              : isInvalidKey

   272→
   273→        // Auto-show config if keys are missing

   276→        }
   277→      }

   284→      };
   285→      setMessages((prev) => [...prev, errorMessage]);

   287→      setIsLoading(false);
   288→    }

   290→

   295→    }

   297→
   298→  const clearChat = () => {
   299→    setMessages([]);
   300→  };
   301→
   302→  // Check if any AI provider is available (Z.AI is always available in sandbox)
   303→  const isAIConfigured = aiConfig?.zai?.available || aiConfig?.gemini?.configured || aiConfig?.groq?.configured;
   304→
   305→  // Format message content with simple markdown-like rendering
   306→  const formatContent = (content: string) => {
   307→    // Split by newlines and handle bullet points
   308→    return content.split('\n').map((line, i) => {
   309→      // Bold text
   310→      const formattedLine = line.replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>');
   311→      // Bullet points
   312→      if (line.trimStart().startsWith('- ') || line.trimStart().startsWith('• ') || line.trimStart().startsWith('* ')) {
   313→        return (
   314→          <div key={i} className="flex gap-1.5 ml-1">
   315→            <span className="shrink-0">•</span>
   316→            <span dangerouslySetInnerHTML={{ __html: formattedLine.replace(/^[\s]*[-•*]\s*/, '') }} />
   317→          </div>
   318→        );
   319→      }
   320→      // Numbered lists
   321→      const numMatch = line.trimStart().match(/^(\d+)\.\s/);
   322→      if (numMatch) {
   323→        return (
   324→          <div key={i} className="flex gap-1.5 ml-1">
   325→            <span className="shrink-0">{numMatch[1]}.</span>
   326→            <span dangerouslySetInnerHTML={{ __html: formattedLine.replace(/^\s*\d+\.\s/, '') }} />
   327→          </div>
   328→        );
   329→      }
   330→      // Empty line = paragraph break
   331→      if (line.trim() === '') {
   332→        return <div key={i} className="h-2" />;
   333→      }
   334→      return <div key={i} dangerouslySetInnerHTML={{ __html: formattedLine }} />;
   335→    });
   336→  };

   342→    if (result === 'testing') return <span className="text-amber-600">⏳ Testing...</span>;
   343→    if (result === 'empty_response') return <span className="text-amber-600">⚠️ {label} kosong</span>;
   344→    return <span className="text-gray-400">⏸ Belum ditest</span>;
   345→  };
   346→
   347→  return (
   348→    <>

   354→            animate={{ scale: 1, opacity: 1 }}
   355→            exit={{ scale: 0, opacity: 0 }}
   356→            transition={{ type: 'spring', stiffness: 260, damping: 20 }}

   360→              background: 'linear-gradient(135deg, #06B6D4, #0891B2)',
   361→              boxShadow: '0 4px 20px rgba(6,182,212,0.4)',

   384→            exit={{ opacity: 0, y: 20, scale: 0.95 }}
   385→            transition={{ type: 'spring', stiffness: 300, damping: 25 }}
   386→            className="fixed bottom-6 right-6 z-50 w-[400px] max-w-[calc(100vw-48px)] flex flex-col rounded-2xl overflow-hidden border"
   387→            style={{
   388→              height: 'min(620px, calc(100vh - 120px))',
   389→              background: 'var(--background)',
   390→              borderColor: 'var(--border)',
   391→              boxShadow: '0 8px 40px rgba(0,0,0,0.15), 0 0 0 1px rgba(6,182,212,0.1)',
   392→            }}
   393→          >
   394→            {/* Header */}
   395→            <div
   396→              className="shrink-0 px-4 py-3 flex items-center gap-3"
   397→              style={{
   398→                background: 'linear-gradient(135deg, #06B6D4, #0891B2)',
   399→              }}
   400→            >
   401→              <div className="w-8 h-8 rounded-lg bg-white/20 flex items-center justify-center">
   402→                <Sparkles className="h-4 w-4 text-white" />
   403→              </div>
   404→              <div className="flex-1 min-w-0">
   405→                <div className="text-sm font-bold text-white">Asisten AI Perikanan</div>
   406→                <div className="text-[10px] text-cyan-100">
   407→                  SIPBD · Mempawah {!isAIConfigured && '· ⚠️ API Key belum diatur'}
   408→                </div>
   409→              </div>
   410→              <button
   411→                onClick={() => { setShowConfig(!showConfig); fetchAIConfig(); }}
   412→                className="w-7 h-7 rounded-lg bg-white/10 flex items-center justify-center hover:bg-white/20 transition-colors"
   413→                title="Pengaturan AI"
   414→              >
   415→                <Settings className="h-3.5 w-3.5 text-white" />
   416→              </button>
   417→              <button
   418→                onClick={clearChat}
   419→                className="w-7 h-7 rounded-lg bg-white/10 flex items-center justify-center hover:bg-white/20 transition-colors"
   420→                title="Hapus percakapan"

   422→                <Trash2 className="h-3.5 w-3.5 text-white" />

   424→              <button
   425→                onClick={() => setIsOpen(false)}

   427→                title="Tutup"
   428→              >
   429→                <X className="h-3.5 w-3.5 text-white" />
   430→              </button>
   431→            </div>
   432→
   433→            {/* Config panel */}
   434→            <AnimatePresence>
   435→              {showConfig && (
   436→                <motion.div
   437→                  initial={{ height: 0, opacity: 0 }}
   438→                  animate={{ height: 'auto', opacity: 1 }}
   439→                  exit={{ height: 0, opacity: 0 }}
   440→                  className="overflow-hidden border-b"
   441→                  style={{ borderColor: 'var(--border)' }}
   442→                >
   443→                  <div className="p-4 space-y-3 max-h-[50vh] overflow-y-auto" style={{ background: 'var(--card)' }}>
   444→                    <div className="text-xs font-semibold flex items-center gap-2" style={{ color: 'var(--foreground)' }}>
   445→                      <Key className="h-3.5 w-3.5" /> Konfigurasi AI Provider
   446→                    </div>
   447→                    <p className="text-[10px]" style={{ color: 'var(--muted-foreground)' }}>
   448→                      Z.AI (chat.z.ai) adalah provider utama — otomatis tersedia, tanpa API key! Gemini/Groq sebagai fallback opsional.
   449→                    </p>
   450→
   451→                    {/* Status indicators */}
   452→                    <div className="flex gap-2 flex-wrap">
   453→                      <div className={`flex items-center gap-1.5 px-2 py-1 rounded-md text-[10px] ${aiConfig?.zai?.available ? 'bg-green-50 text-green-700' : 'bg-amber-50 text-amber-700'}`}>
   454→                        {aiConfig?.zai?.available ? <CheckCircle2 className="h-3 w-3" /> : <AlertCircle className="h-3 w-3" />}
   455→                        Z.AI {aiConfig?.zai?.available ? '✓' : '✗'} <span className="opacity-60">(primary)</span>
   456→                      </div>
   457→                      <div className={`flex items-center gap-1.5 px-2 py-1 rounded-md text-[10px] ${aiConfig?.gemini?.configured ? 'bg-green-50 text-green-700' : 'bg-gray-50 text-gray-500'}`}>

   459→                        Gemini {aiConfig?.gemini?.configured ? '✓' : '✗'}
   460→                        {aiConfig?.gemini?.keyHint && <span className="opacity-60">({aiConfig.gemini.keyHint})</span>}
   461→                      </div>
   462→                      <div className={`flex items-center gap-1.5 px-2 py-1 rounded-md text-[10px] ${aiConfig?.groq?.configured ? 'bg-green-50 text-green-700' : 'bg-gray-50 text-gray-500'}`}>
   463→                        {aiConfig?.groq?.configured ? <CheckCircle2 className="h-3 w-3" /> : <AlertCircle className="h-3 w-3" />}
   464→                        Groq {aiConfig?.groq?.configured ? '✓' : '✗'}
   465→                        {aiConfig?.groq?.keyHint && <span className="opacity-60">({aiConfig.groq.keyHint})</span>}
   466→                      </div>
   467→                    </div>
   468→
